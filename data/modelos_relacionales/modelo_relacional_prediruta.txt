Table ACCIDENTE {
  ACCIDENTE_ID integer [pk]
  FECHA_HORA timestamp [not null]
  LATITUD decimal(9,6) [not null]
  LONGITUD decimal(9,6) [not null]
  CLASE_ACCIDENTE varchar
  CLIMA_ID integer

  OBJETIVO_GRAVE boolean [not null]

  Note: 'Una fila por accidente válido dentro del alcance definido para PrediRuta.'
}

Table VIA {
  ACCIDENTE_ID integer [pk]
  GEOMETRIA_PLANTA varchar
  GEOMETRIA_TERRENO varchar
  GEOMETRIA_SECCION varchar
  SENTIDO_VIA varchar
  N_CALZADAS integer
  N_CARRILES integer
  SUPERFICIE_RODADURA varchar
  ESTADO_VIA varchar
  CONDICION_VIA varchar
  ILUMINACION_ARTIFICIAL varchar
  SEMAFORO varchar

  Note: 'Una vía representativa por accidente, seleccionada mediante una regla determinística.'
}

Table CLIMA {
  CLIMA_ID integer [pk]
  FECHA_HORA_CLIMA timestamp [not null]

  TEMPERATURA_2M decimal
  HUMEDAD_RELATIVA_2M decimal
  SENSACION_TERMICA decimal
  PRECIPITACION decimal
  LLUVIA decimal
  NUBOSIDAD decimal
  PRESION_SUPERFICIE decimal
  VELOCIDAD_VIENTO_10M decimal
  DIRECCION_VIENTO_10M decimal

  Note: 'Una observación meteorológica única relacionada con ACCIDENTE mediante CLIMA_ID.'
}

Table VEHICULO {
  ACCIDENTE_ID integer [not null]
  VEHICULO_ID integer [pk]
  CLASE varchar
  SERVICIO varchar

  Note: 'Vehículos asociados a los accidentes utilizables. La placa no se conserva.'
}

Table ACTOR_VIAL {
  ACCIDENTE_ID integer [not null]
  ACTOR_ID integer [pk]
  VEHICULO_ID integer
  CONDICION varchar
  ESTADO varchar
  GENERO varchar
  EDAD integer

  Note: 'Actores asociados a los accidentes utilizables de forma anonimizada.'
}

Table CAUSA {
  CAUSA_ID integer [pk]
  ACCIDENTE_ID integer [not null]
  VEHICULO_ID integer
  CODIGO_CAUSA varchar [not null]
  NOMBRE varchar
  TIPO varchar

  Note: 'Causas asociadas al accidente o a uno de sus vehículos.'
}
